# Dependencies for the RootScope v4 Hugging Face Space.
#
# The Space vendors its own copy of rootscope/ (uploaded next to app.py), so
# RootScope is NOT installed from GitHub here. Doing both would put two
# packages of the same name on sys.path and whichever imports first wins --
# a bug that shows up only as wrong predictions.
#
# torch is NOT pinned, deliberately: ZeroGPU supports only torch >= 2.8 and
# picks the version. torch 2.6+ defaults torch.load(weights_only=True), which
# backbone.pt and the Cellpose-SAM checkpoint must load under -- test the two
# example images on the Space before announcing a change.
#
# v4 needs no xgboost: the classifier is LightGBM only. It is not listed on
# purpose (unpickling an XGBoost Booster is what poisons ZeroGPU's CUDA state;
# see classify_worker.py). Do not add it back for "completeness".

gradio>=5.0
spaces>=0.30

# scikit-learn is a hard pin: the LightGBM bundles carry StandardScalers pickled
# with 1.7.2, and other versions can fail to unpickle or change results.
scikit-learn==1.7.2

numpy>=1.24
